set(probe_sources)
if(CONFIG_FTM_RESPONDER_PROBE)
    set(blob_dir "${CMAKE_CURRENT_BINARY_DIR}/blob")
    file(MAKE_DIRECTORY "${blob_dir}")
    foreach(object IN ITEMS ieee80211_ftm wdev lmac)
        if(object STREQUAL "ieee80211_ftm")
            set(archive libnet80211.a)
            set(expected_hash "79607f4b192e89b4e2380ec2a332c6d64907af49dfbb8d63f8f7ac901fee4500")
        elseif(object STREQUAL "lmac")
            set(archive libpp.a)
            set(expected_hash "93bd0248a9a413ab1789c5a859b187763f2ed47afd34c80dda09b265941d6947")
        else()
            set(archive libpp.a)
            set(expected_hash "1936550cb69bc8c9f9c92e642751aa42532270e94c0e0e082daea9720cbbe81c")
        endif()
        execute_process(COMMAND ${CMAKE_AR} x "${IDF_PATH}/components/esp_wifi/lib/esp32c6/${archive}" "${object}.o" WORKING_DIRECTORY "${blob_dir}" COMMAND_ERROR_IS_FATAL ANY)
        file(SHA256 "${blob_dir}/${object}.o" object_hash)
        if(NOT object_hash STREQUAL expected_hash)
            message(FATAL_ERROR "FTM responder probe requires the audited ${object} object")
        endif()
    endforeach()
    list(APPEND probe_sources "ftm_responder_probe.c")
endif()
idf_component_register(SRCS ${probe_sources} PRIV_REQUIRES esp_wifi esp_timer WHOLE_ARCHIVE)
if(CONFIG_FTM_RESPONDER_PROBE)
    target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,--wrap=wDev_ftm_record_t1t4")
endif()
